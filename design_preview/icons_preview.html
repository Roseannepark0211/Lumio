Usage
<!-- 1. In HTML — reference sprite via <use> -->
<svg class="icon" viewBox="0 0 24 24">
  <use href="icons.svg#i-home"/>
</svg>

/* 2. In CSS — color inherits from parent */
.icon {
  width: 24px;
  height: 24px;
  color: var(--text-primary);
  /* hover/active states just change color */
}
.icon:hover { color: var(--accent); }
